type Case struct {
  ID       string `json:"id"`
  Seam     string `json:"seam"`
  Input    any    `json:"input"`
  Expected any    `json:"expected"`
}


// qs quotes a whole JSON string; qo opens one and qc closes one. Splitting the
// quoting lets the two case-folding probes below be assembled from runes
// rather than from escape sequences.
func qs(s string) string { return string(rune(34)) + s + string(rune(34)) }
func qo(s string) string { return string(rune(34)) + s }
func qc(s string) string { return s + string(rune(34)) }

// kelvin is U+212A KELVIN SIGN, which Unicode simple-folds onto "k".
func kelvin() string { return string(rune(0x212A)) }

// longS is U+017F LATIN SMALL LETTER LONG S, which simple-folds onto "s".
func longS() string { return string(rune(0x017F)) }

func fail(err error) {
  if err != nil {
    panic(err)
  }
}

func b64(b []byte) string { return base64.StdEncoding.EncodeToString(b) }

func unb64(s string) []byte {
  b, e := base64.StdEncoding.DecodeString(s)
  fail(e)
  return b
}

func hash(b []byte) string { s := sha256.Sum256(b); return hex.EncodeToString(s[:]) }

func mustTime(s string) time.Time {
  t, e := time.Parse(time.RFC3339Nano, s)
  fail(e)
  return t
}

// marshalCase freezes the exact bytes json.MarshalIndent produces for one
// handshake artifact, including Go's HTML escaping and RFC3339Nano time shape.
func marshalPairing(id, token, publicKey, createdAt string) Case {
  pf := pairing.PairingFile{Token: token, PublicKey: publicKey, CreatedAt: mustTime(createdAt)}
  data, e := pairing.MarshalPairingFile(pf)
  fail(e)
  return Case{
    ID:   id,
    Seam: "PAIRING-001",
    Input: map[string]any{
      "kind": "pairing_file", "token": token, "public_key": publicKey, "created_at": createdAt,
    },
    Expected: map[string]any{"bytes_b64": b64(data), "sha256": hash(data)},
  }
}

// JoinResponse has no exported marshaller in the pinned oracle; cmd/device.go
// writes it through savePairingFile, which is json.MarshalIndent with the same
// two-space indent. Freeze that exact call so the Rust encoder is bound to the
// real production shape rather than to a re-implementation.
func marshalJoin(id, token, name, publicKey, createdAt string) Case {
  jr := pairing.JoinResponse{Token: token, Name: name, PublicKey: publicKey, CreatedAt: mustTime(createdAt)}
  data, e := json.MarshalIndent(jr, "", "  ")
  fail(e)
  return Case{
    ID:   id,
    Seam: "PAIRING-001",
    Input: map[string]any{
      "kind": "join_response", "token": token, "name": name, "public_key": publicKey, "created_at": createdAt,
    },
    Expected: map[string]any{"bytes_b64": b64(data), "sha256": hash(data)},
  }
}

// createdAt reports how Go re-serializes a parsed timestamp. Go's RFC3339
// parser is more permissive than its marshaller — "+24:00" parses but cannot
// be marshalled — so the marshal error is part of the frozen contract.
func createdAt(t time.Time) map[string]any {
  created, e := t.MarshalJSON()
  if e != nil {
    return map[string]any{"created_at_marshal_error": e.Error()}
  }
  return map[string]any{"created_at_json": string(created)}
}

func parsedPairing(pf pairing.PairingFile) map[string]any {
  out := map[string]any{"token": pf.Token, "public_key": pf.PublicKey}
  for k, v := range createdAt(pf.CreatedAt) {
    out[k] = v
  }
  return out
}

func parsedJoin(jr pairing.JoinResponse) map[string]any {
  out := map[string]any{"token": jr.Token, "name": jr.Name, "public_key": jr.PublicKey}
  for k, v := range createdAt(jr.CreatedAt) {
    out[k] = v
  }
  return out
}

func parsePairingCase(id, raw string) Case {
  pf, err := pairing.ParsePairingFile([]byte(raw))
  expected := map[string]any{"ok": err == nil}
  if err == nil {
    expected["parsed"] = parsedPairing(pf)
  } else {
    expected["go_error"] = err.Error()
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"kind": "pairing_file", "data_b64": b64([]byte(raw))},
    Expected: expected}
}

func parseJoinCase(id, raw string) Case {
  jr, err := pairing.ParseJoinResponse([]byte(raw))
  expected := map[string]any{"ok": err == nil}

// storeCase replays an operation script against a real pairing.TokenStore.
// Expiry is expressed through the exported pairing.TokenTTL rather than by
// sleeping, so the frozen expectations are wall-clock independent and the
// --check re-run is not timing dependent.
func storeCase(id string, ops []map[string]any) Case {
  original := pairing.TokenTTL
  defer func() { pairing.TokenTTL = original }()
  ts := pairing.NewTokenStore()
  results := make([]map[string]any, 0, len(ops))
  for _, op := range ops {
    switch op["op"].(string) {
    case "set_ttl_ms":
      pairing.TokenTTL = time.Duration(int64(op["ttl_ms"].(int))) * time.Millisecond
      results = append(results, map[string]any{"op": "set_ttl_ms"})
    case "store":
      e := ts.Store(pairing.Token(op["token"].(string)), op["public_key"].(string))
      results = append(results, map[string]any{"op": "store", "ok": e == nil})
    case "validate":
      key, ok := ts.Validate(op["token"].(string))
      results = append(results, map[string]any{"op": "validate", "ok": ok, "public_key": key})
    case "cleanup":
      ts.CleanupExpired()
      results = append(results, map[string]any{"op": "cleanup"})
    default:
      panic("unknown op " + op["op"].(string))
    }
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"ops": ops},
    Expected: map[string]any{"results": results}}
}

func op(kind string, kv ...any) map[string]any {
  m := map[string]any{"op": kind}
  for i := 0; i+1 < len(kv); i += 2 {
    m[kv[i].(string)] = kv[i+1]
  }
  return m
}


// ---- device registry (devices.json) -------------------------------------
//
// The registry is the half of PAIRING-001 that `device add`, `device list` and
// `device revoke` operate on. Every case runs against a fresh vault directory
// so nothing carries over.

func mustTimePtr(s string) *time.Time {
  if s == "" {
    return nil
  }
  t := mustTime(s)
  return &t
}

func registryFile(vaultDir string) map[string]any {
  path := filepath.Join(vaultDir, config.DefaultVaultSubdir, "devices.json")
  data, err := os.ReadFile(path)
  if err != nil {
    if os.IsNotExist(err) {
      return map[string]any{"exists": false}
    }
    panic(err)
  }
  return map[string]any{"exists": true, "bytes_b64": b64(data), "sha256": hash(data)}
}

func devicesJSON(devices []vault.Device) string {
  data, e := json.Marshal(devices)
  fail(e)
  return string(data)
}

// registryCase replays an operation script against the real DeviceManager.
// Only platform-independent output is recorded here: JSON bytes, returned
// values and whether an operation failed. Unix file modes are a separate
// case group, because Windows reports POSIX bits that carry no meaning.
func registryCase(id string, ops []map[string]any) Case {
  dir, e := os.MkdirTemp("", "pairing-registry-")
  fail(e)
  defer func() { _ = os.RemoveAll(dir) }()
  dm := vault.NewDeviceManager(dir)
  results := make([]map[string]any, 0, len(ops))
  for _, o := range ops {
    switch o["op"].(string) {
    case "add":
      lastSeen := ""
      if v, ok := o["last_seen"].(string); ok {
        lastSeen = v
      }
      err := dm.AddDevice(vault.Device{
        Name:      o["name"].(string),
        PublicKey: o["public_key"].(string),
        AddedAt:   mustTime(o["added_at"].(string)),
        LastSeen:  mustTimePtr(lastSeen),
      })
      results = append(results, map[string]any{"op": "add", "ok": err == nil})
    case "remove":
      err := dm.RemoveDevice(o["name"].(string))
      results = append(results, map[string]any{"op": "remove", "ok": err == nil})
    case "get":
      device, err := dm.GetDevice(o["name"].(string))
      entry := map[string]any{"op": "get", "ok": err == nil, "found": device != nil}
      if device != nil {
        entry["device"] = devicesJSON([]vault.Device{*device})
      }
      results = append(results, entry)
    case "list":
      devices, err := dm.ListDevices()
      entry := map[string]any{"op": "list", "ok": err == nil}
      if err == nil {
        entry["devices"] = devicesJSON(devices)
      }
      results = append(results, entry)
    case "corrupt":
      path := filepath.Join(dir, config.DefaultVaultSubdir, "devices.json")
      fail(os.MkdirAll(filepath.Dir(path), 0o700))
      fail(os.WriteFile(path, unb64(o["data_b64"].(string)), 0o600))
      results = append(results, map[string]any{"op": "corrupt"})
    case "file":
      entry := registryFile(dir)
      entry["op"] = "file"
      results = append(results, entry)
    default:
      panic("unknown registry op " + o["op"].(string))
    }
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"ops": ops},
    Expected: map[string]any{"results": results}}
}

// registryModesCase records the permission bits the registry creates. It is a
// separate group because Windows has no POSIX modes to compare; the generator
// skips this group's payload there rather than freezing a meaningless value.
func registryModesCase(id string) Case {
  dir, e := os.MkdirTemp("", "pairing-registry-modes-")
  fail(e)
  defer func() { _ = os.RemoveAll(dir) }()
  dm := vault.NewDeviceManager(dir)
  fail(dm.AddDevice(vault.Device{Name: "laptop", PublicKey: "age1modes", AddedAt: mustTime("2026-09-14T18:45:00Z")}))
  subdir := filepath.Join(dir, config.DefaultVaultSubdir)
  path := filepath.Join(subdir, "devices.json")
  dirInfo, e := os.Stat(subdir)
  fail(e)
  fileInfo, e := os.Stat(path)
  fail(e)
  return Case{ID: id, Seam: "PAIRING-001",
    Input: map[string]any{"operation": "AddDevice into an empty vault directory"},
    Expected: map[string]any{
      "dir_mode":  fmt.Sprintf("%04o", dirInfo.Mode().Perm()),
      "file_mode": fmt.Sprintf("%04o", fileInfo.Mode().Perm()),
      "goos":      runtime.GOOS,
    }}
}

// registryLayoutCase observes where the registry is created rather than
// re-deriving the path: everything under the vault directory is walked after a
// single AddDevice and recorded as slash-joined relative paths, so the layout
// contract holds on every platform and a renamed directory or file is caught.
func registryLayoutCase(id string) Case {
  dir, e := os.MkdirTemp("", "pairing-registry-layout-")
  fail(e)
  defer func() { _ = os.RemoveAll(dir) }()
  dm := vault.NewDeviceManager(dir)
  fail(dm.AddDevice(vault.Device{Name: "laptop", PublicKey: "age1layout", AddedAt: mustTime("2026-09-14T18:45:00Z")}))
  var entries []string
  fail(filepath.Walk(dir, func(p string, info os.FileInfo, err error) error {
    fail(err)
    if p == dir {
      return nil
    }
    rel, relErr := filepath.Rel(dir, p)
    fail(relErr)
    kind := "file"
    if info.IsDir() {
      kind = "dir"
    }
    entries = append(entries, kind+":"+filepath.ToSlash(rel))
    return nil
  }))
  sort.Strings(entries)
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"operation": "AddDevice into an empty vault directory"},
    Expected: map[string]any{"entries": entries}}
}
