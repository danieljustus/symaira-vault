type Case struct {
  ID       string `json:"id"`
  Seam     string `json:"seam"`
  Input    any    `json:"input"`
  Expected any    `json:"expected"`
}

func fail(err error) {
  if err != nil {
    panic(err)
  }
}

func b64(b []byte) string { return base64.StdEncoding.EncodeToString(b) }

func unb64(s string) []byte {
  b, e := base64.StdEncoding.DecodeString(s)
  fail(e)
  return b
}

func hash(b []byte) string { s := sha256.Sum256(b); return hex.EncodeToString(s[:]) }

func mustTime(s string) time.Time {
  t, e := time.Parse(time.RFC3339Nano, s)
  fail(e)
  return t
}

// marshalCase freezes the exact bytes json.MarshalIndent produces for one
// handshake artifact, including Go's HTML escaping and RFC3339Nano time shape.
func marshalPairing(id, token, publicKey, createdAt string) Case {
  pf := pairing.PairingFile{Token: token, PublicKey: publicKey, CreatedAt: mustTime(createdAt)}
  data, e := pairing.MarshalPairingFile(pf)
  fail(e)
  return Case{
    ID:   id,
    Seam: "PAIRING-001",
    Input: map[string]any{
      "kind": "pairing_file", "token": token, "public_key": publicKey, "created_at": createdAt,
    },
    Expected: map[string]any{"bytes_b64": b64(data), "sha256": hash(data)},
  }
}

// JoinResponse has no exported marshaller in the pinned oracle; cmd/device.go
// writes it through savePairingFile, which is json.MarshalIndent with the same
// two-space indent. Freeze that exact call so the Rust encoder is bound to the
// real production shape rather than to a re-implementation.
func marshalJoin(id, token, name, publicKey, createdAt string) Case {
  jr := pairing.JoinResponse{Token: token, Name: name, PublicKey: publicKey, CreatedAt: mustTime(createdAt)}
  data, e := json.MarshalIndent(jr, "", "  ")
  fail(e)
  return Case{
    ID:   id,
    Seam: "PAIRING-001",
    Input: map[string]any{
      "kind": "join_response", "token": token, "name": name, "public_key": publicKey, "created_at": createdAt,
    },
    Expected: map[string]any{"bytes_b64": b64(data), "sha256": hash(data)},
  }
}

// createdAt reports how Go re-serializes a parsed timestamp. Go's RFC3339
// parser is more permissive than its marshaller — "+24:00" parses but cannot
// be marshalled — so the marshal error is part of the frozen contract.
func createdAt(t time.Time) map[string]any {
  created, e := t.MarshalJSON()
  if e != nil {
    return map[string]any{"created_at_marshal_error": e.Error()}
  }
  return map[string]any{"created_at_json": string(created)}
}

func parsedPairing(pf pairing.PairingFile) map[string]any {
  out := map[string]any{"token": pf.Token, "public_key": pf.PublicKey}
  for k, v := range createdAt(pf.CreatedAt) {
    out[k] = v
  }
  return out
}

func parsedJoin(jr pairing.JoinResponse) map[string]any {
  out := map[string]any{"token": jr.Token, "name": jr.Name, "public_key": jr.PublicKey}
  for k, v := range createdAt(jr.CreatedAt) {
    out[k] = v
  }
  return out
}

func parsePairingCase(id, raw string) Case {
  pf, err := pairing.ParsePairingFile([]byte(raw))
  expected := map[string]any{"ok": err == nil}
  if err == nil {
    expected["parsed"] = parsedPairing(pf)
  } else {
    expected["go_error"] = err.Error()
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"kind": "pairing_file", "data_b64": b64([]byte(raw))},
    Expected: expected}
}

func parseJoinCase(id, raw string) Case {
  jr, err := pairing.ParseJoinResponse([]byte(raw))
  expected := map[string]any{"ok": err == nil}
  if err == nil {
    expected["parsed"] = parsedJoin(jr)
  } else {
    expected["go_error"] = err.Error()
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"kind": "join_response", "data_b64": b64([]byte(raw))},
    Expected: expected}
}

func filenamesCase(id, token string) Case {
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"token_b64": b64([]byte(token))},
    Expected: map[string]any{"names": pairing.ResponseFilenames(token)}}
}

func validateCase(id, token string) Case {
  err := pairing.ValidatePairingToken(token)
  expected := map[string]any{"ok": err == nil}
  if err != nil {
    expected["go_error"] = err.Error()
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"token_b64": b64([]byte(token))},
    Expected: expected}
}

func displayCase(id, token string) Case {
  return Case{ID: id, Seam: "PAIRING-001",
    Input: map[string]any{"token_b64": b64([]byte(token))},
    Expected: map[string]any{
      "display": pairing.Token(token).Display(),
      "string":  pairing.Token(token).String(),
    }}
}

// storeCase replays an operation script against a real pairing.TokenStore.
// Expiry is expressed through the exported pairing.TokenTTL rather than by
// sleeping, so the frozen expectations are wall-clock independent and the
// --check re-run is not timing dependent.
func storeCase(id string, ops []map[string]any) Case {
  original := pairing.TokenTTL
  defer func() { pairing.TokenTTL = original }()
  ts := pairing.NewTokenStore()
  results := make([]map[string]any, 0, len(ops))
  for _, op := range ops {
    switch op["op"].(string) {
    case "set_ttl_ms":
      pairing.TokenTTL = time.Duration(int64(op["ttl_ms"].(int))) * time.Millisecond
      results = append(results, map[string]any{"op": "set_ttl_ms"})
    case "store":
      e := ts.Store(pairing.Token(op["token"].(string)), op["public_key"].(string))
      results = append(results, map[string]any{"op": "store", "ok": e == nil})
    case "validate":
      key, ok := ts.Validate(op["token"].(string))
      results = append(results, map[string]any{"op": "validate", "ok": ok, "public_key": key})
    case "cleanup":
      ts.CleanupExpired()
      results = append(results, map[string]any{"op": "cleanup"})
    default:
      panic("unknown op " + op["op"].(string))
    }
  }
  return Case{ID: id, Seam: "PAIRING-001",
    Input:    map[string]any{"ops": ops},
    Expected: map[string]any{"results": results}}
}

func op(kind string, kv ...any) map[string]any {
  m := map[string]any{"op": kind}
  for i := 0; i+1 < len(kv); i += 2 {
    m[kv[i].(string)] = kv[i+1]
  }
  return m
}
