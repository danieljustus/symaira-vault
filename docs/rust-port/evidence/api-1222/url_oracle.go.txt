package main

import (
	"encoding/json"
	"net/url"
	"os"
)

type vector struct {
	Value   string `json:"value"`
	Query   string `json:"query"`
	Segment string `json:"segment"`
	Path    string `json:"path"`
}

func main() {
	var input []vector
	if err := json.NewDecoder(os.Stdin).Decode(&input); err != nil {
		panic(err)
	}
	for i := range input {
		input[i].Query = url.QueryEscape(input[i].Value)
		input[i].Segment = url.PathEscape(input[i].Value)
		input[i].Path = (&url.URL{Path: input[i].Value}).EscapedPath()
	}
	if err := json.NewEncoder(os.Stdout).Encode(input); err != nil {
		panic(err)
	}
}
